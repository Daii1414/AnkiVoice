# kana_overrides.txt — словарь замен перед синтезом.
# Строки вида: иероглифы=читается_как
# Применяются к тексту карточек ДО перевода в хирагану.
# Нужно только там, где MeCab (pyopenjtalk) читает слово НЕВЕРНО —
# лишние замены не только излишни, они меняют контекст разбора
# и могут сломать чтение соседних слов (проверено опытным путём).
# Длинные совпадения применяются первыми, порядок строк не важен.

# --- архитектурные термины: MeCab даёт неверное чтение ---
合折釘=あいおれくぎ
合釘=あいくぎ
合印=あいじるし
合端=あいば
相欠き=あいがき
相欠き継ぎ=あいがきつぎ
相决り=あいじゃくり
相决り継ぎ=あいじゃくりつぎ
相决り矧ぎ=あいじゃくりはぎ
相の間=あいのまい
石の間=いしのまい
間の宿=あいのしゅく
板矧ぎ=いたばぎ
野地板=のじいた
仕口=しくち
権現造=ごんげんづくり
本宿=ほんしゅく

# --- приветствия: MeCab читает хирагану «こんにちは» как частицу с «わ» ---
# Пишем КАТАКАНОЙ — её MeCab разбирает как обычное слово и читает верно.
# kanaize потом вернёт хирагану там, где это безопасно, а эти слова оставит.
こんにちは=コンニチハ
こんばんは=コンバンハ

# --- иероглифы, которых нет в словаре MeCab (пропускаются в текст «как есть») ---

# --- слова, которые MeCab читает с оговорками (на всякий случай оставлены) ---

# --- латиница в словах (катакана) ---
